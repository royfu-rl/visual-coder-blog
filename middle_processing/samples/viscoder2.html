import seaborn as sns
import matplotlib.pyplot as plt

# Set the style
sns.set(style="whitegrid")

# Load the Titanic dataset
titanic = sns.load_dataset("titanic")

# Create a grouped barplot
g = sns.catplot(x="class", y="survived", hue="sex",
                data=titanic, kind="bar",
                height=6, palette="muted")

# Customize the plot
g.despine(left=True)
g.set_ylabels("survival probability")

# Show the plot
plt.show()

\documentclass[border=10pt]{standalone}
\usepackage[siunitx, RPvoltages]{circuitikz}
\usepackage{tikz}

\begin{document}
\begin{circuitikz}[scale=1.2]
    \draw (0,0) node[left]{Input} to[short] (1,0);
    \draw (1,0) to[R, l=$R$, i=$i$] (4,0);
    \draw (4,0) to[short] (5,0) to[short] (5,-2) to[C, l=$C$] (1,-2) to[short] (1,0);
    \draw (5,0) to[short] (6,0) node[right]{Output};
    
    \draw[thick,->] (2.5,-2.5) -- (2.5,-1) node[midway,right,rotate=90] {Charge};
    \draw[thick,->] (3.5,-1) -- (3.5,-2.5) node[midway,left,rotate=90] {Discharge};
    
    \node[text width=4cm, align=center] at (3,-3.5) {\small RC Circuit for\\Envelope Generator\\in Funk Bass Synth};
\end{circuitikz}
\end{document}
